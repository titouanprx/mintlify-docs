---
title: 'Get Account Details'
api: 'GET /v2/accounts/{account_id}'
description: 'Get detailed information about a specific account'
---

Retrieve detailed information about a specific account, including its configuration and current status.

<Note>
  This endpoint costs **0 credits**.
</Note>

### Header Parameters

<ParamField header="x-api-key" type="string" required>
  Your API key
</ParamField>

### Path Parameters

<ParamField path="account_id" type="string" required>
  The unique identifier of the account to retrieve
</ParamField>

### Response

<ResponseField name="success" type="boolean">
  Whether the request was successful
</ResponseField>

<ResponseField name="data" type="object">
  <Expandable title="Properties">
    <ResponseField name="account_id" type="string">
      Unique identifier for the account
    </ResponseField>

    <ResponseField name="platform" type="string">
      Platform type (e.g. "linkedin")
    </ResponseField>

    <ResponseField name="account_name" type="string">
      Display name of the account
    </ResponseField>

    <ResponseField name="is_active" type="boolean">
      Whether the account is currently active and usable
    </ResponseField>

    <ResponseField name="status" type="string">
      Account status, always in step with `is_active`:

          - `connected` — ready to use (`is_active: true`)
          - `checkpoint_required` — waiting on a 2FA code, see [Checkpoint](/api-reference/v2/accounts/checkpoint)
          - `disconnected` — the LinkedIn session expired; re-authenticate via [Connect Account](/api-reference/v2/accounts/login)
    </ResponseField>

    <ResponseField name="country" type="string">
      Country code associated with the account
    </ResponseField>

    <ResponseField name="created_at" type="string">
      ISO 8601 timestamp of when the account was created
    </ResponseField>

    <ResponseField name="updated_at" type="string">
      ISO 8601 timestamp of the last account update
    </ResponseField>

    <ResponseField name="sales_nav" type="object">
      LinkedIn accounts only — state of the account's Sales Nav seat, same shape as in [List Accounts](/api-reference/v2/accounts/list-accounts#sales-nav-seat). Email accounts return their mailbox fields instead (`from_email`, `provider`, `esp`, `auth_mode`, `daily_limit`, `sends_today`, `sends_total`, `bounce_rate`).
      <Expandable title="sales_nav properties">
        <ResponseField name="connected" type="boolean">
          Whether the account currently has a usable Sales Nav seat.
        </ResponseField>
        <ResponseField name="contract_id" type="string">
          Sales Nav contract the seat belongs to.
        </ResponseField>
        <ResponseField name="seat_id" type="string">
          The seat within that contract.
        </ResponseField>
        <ResponseField name="connected_at" type="string">
          Last seat activation or refresh. Only when `connected` is `true`.
        </ResponseField>
        <ResponseField name="reason" type="string">
          Why the seat is unavailable. Only when `connected` is `false` and a seat existed before.
        </ResponseField>
        <ResponseField name="disconnected_at" type="string">
          When the seat was lost. Only when the seat died on its own.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="metadata" type="object">
  <Expandable title="Properties">
    <ResponseField name="action" type="string">
      The action performed ("get_account")
    </ResponseField>

    <ResponseField name="account_id" type="string">
      The account_id that was queried
    </ResponseField>

    <ResponseField name="credits_consumed" type="integer">
      Number of credits consumed (always 0 for this endpoint)
    </ResponseField>

    <ResponseField name="timestamp" type="string">
      ISO 8601 timestamp of the response
    </ResponseField>
  </Expandable>
</ResponseField>

### Notes
- This endpoint is free and does not consume any credits
