---
title: 'Search Companies'
api: 'POST /v2/profiles'
description: 'Search for company pages using various filters.'
---

## Overview

Search for company pages using keyword, location, sector, and size filters. Supports pagination for retrieving large result sets.

<Note>This endpoint consumes **1 credit per 10 results** returned.</Note>

## Request

<ParamField header="x-api-key" type="string" required>
  Your API key for authentication.
</ParamField>

<ParamField body="account_id" type="string" required>
  The unique identifier of the account to use.
</ParamField>

<ParamField body="action" type="string" required>
  Must be `"search_companies"`.
</ParamField>

<ParamField body="params" type="object">
  <Expandable title="params properties">
    <ParamField body="keyword" type="string">
      General keyword to search for across company profiles.
    </ParamField>
    <ParamField body="location" type="string | string[]">
      Filter by company location. Accepts a single string or an array (e.g., `["New York", "Germany"]`).
    </ParamField>
    <ParamField body="sector" type="string | string[]">
      Filter by industry sector. Accepts a single string or an array.
    </ParamField>
    <ParamField body="company_size" type="string | string[]">
      Filter by company size range. Accepts a single string or an array (e.g., `["11-50", "201-500"]`).
    </ParamField>
    <ParamField body="offset" type="integer" default={0}>
      Index of the first result to return (0-based). Combine with `count` to paginate. Any value works — it does not have to align to a page boundary.
    </ParamField>
    <ParamField body="count" type="integer" default={10}>
      Number of results to return, starting at `offset`.
    </ParamField>
  </Expandable>
</ParamField>


## Response

<ResponseField name="success" type="boolean">
  Indicates whether the request was successful.
</ResponseField>

<ResponseField name="data" type="object">
  <Expandable title="data properties">
    <ResponseField name="companies" type="array">
      Matching companies.
      <Expandable title="company properties">
        <ResponseField name="company_name" type="string">Company name.</ResponseField>
        <ResponseField name="company_url" type="string">LinkedIn company page URL.</ResponseField>
        <ResponseField name="industry" type="string">Industry, e.g. `"IT Services and IT Consulting"`.</ResponseField>
        <ResponseField name="location" type="string">Headquarters location.</ResponseField>
        <ResponseField name="logo_url" type="string">Company logo URL.</ResponseField>
        <ResponseField name="subscribers" type="string">Follower count as displayed, e.g. `"114K followers"`.</ResponseField>
        <ResponseField name="subscribers_count" type="integer">Same count as a number.</ResponseField>
        <ResponseField name="job_offers" type="string">Open jobs as displayed, `null` when LinkedIn shows none.</ResponseField>
        <ResponseField name="job_offers_count" type="integer">Same count as a number, `null` when unavailable.</ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="total_results" type="integer">
      Number of companies returned.
    </ResponseField>
    <ResponseField name="total_available_results" type="integer">
      Total number of matching companies available.
    </ResponseField>
  </Expandable>
</ResponseField>
