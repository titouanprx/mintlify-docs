---
title: 'Get Profile Info'
api: 'POST /v2/profiles'
description: 'Retrieve the profile information of any user.'
---

## Overview

Fetches detailed profile data for a specified user. You can identify the target profile using a profile URL, a public identifier, or a profile URN.

<Note>This endpoint consumes **1 credit** per request.</Note>

## Request

<ParamField header="x-api-key" type="string" required>
  Your API key for authentication.
</ParamField>

<ParamField body="account_id" type="string" required>
  The unique identifier of the account to use.
</ParamField>

<ParamField body="action" type="string" required>
  Must be `"get"`.
</ParamField>

<ParamField body="params" type="object">
  <Expandable title="params properties">
    <ParamField body="profile_url" type="string" required>
      Profile URL. Accepts a classic URL (`/in/...`) or a Sales Nav lead URL (`/sales/lead/...`). Required if `identifier` and `profile_urn` are not provided.
    </ParamField>
    <ParamField body="identifier" type="string">
      Public identifier (vanity name). Alternative to `profile_url`.
    </ParamField>
    <ParamField body="profile_urn" type="string">
      Profile URN. Alternative to `profile_url`.
    </ParamField>
    <ParamField body="sales_nav" type="boolean" default={false}>
      Also return `open_link` (Open Profile) and `inmail_eligible`. Requires a Sales Navigator seat. Auto-enabled for a Sales Nav lead URL.
    </ParamField>
  </Expandable>
</ParamField>


## Response

<ResponseField name="success" type="boolean">
  Indicates whether the request was successful.
</ResponseField>

<ResponseField name="data" type="object">
  The profile data of the target user.
  <Expandable title="data properties">
    <ResponseField name="public_id" type="string">Public identifier (vanity name).</ResponseField>
    <ResponseField name="profile_urn" type="string">Unique identifier of the profile. Treat it as an opaque string and pass it back exactly as returned.</ResponseField>
    <ResponseField name="profile_url" type="string">Canonical profile URL.</ResponseField>
    <ResponseField name="first_name" type="string">First name.</ResponseField>
    <ResponseField name="last_name" type="string">Last name.</ResponseField>
    <ResponseField name="headline" type="string">Profile headline.</ResponseField>
    <ResponseField name="location" type="string">Location text.</ResponseField>
    <ResponseField name="summary" type="string">About / bio section.</ResponseField>
    <ResponseField name="industry" type="string">Industry associated with the profile.</ResponseField>
    <ResponseField name="experience" type="array">
      Work experience entries. Each item: `company`, `title`, `description`, `start_date` and `end_date` formatted as `M/YYYY` (e.g. `"10/2025"`; `end_date` is `""` for current roles), `company_url`, `company_logo_url`.
    </ResponseField>
    <ResponseField name="education" type="array">
      Education entries. Each item: `school`, `degree`, `field_of_study`, `start_date`, `end_date` (years, e.g. `"2018"`), `school_logo_url`.
    </ResponseField>
    <ResponseField name="skills" type="array">List of skills (array of strings).</ResponseField>
    <ResponseField name="profile_picture_url" type="string">URL of the profile picture.</ResponseField>
    <ResponseField name="isOpenToWork" type="boolean">Profile displays the "Open to work" badge.</ResponseField>
    <ResponseField name="isHiring" type="boolean">Profile displays the "Hiring" badge.</ResponseField>
    <ResponseField name="relationship" type="string">
      Network distance between the connected account and this profile: `"1st degree"`, `"2nd degree"`, `"3rd+ degree"` or `"Out of network"`. Same vocabulary as `connection_level` in [Search People](/api-reference/v2/profiles/search-people).
    </ResponseField>
    <ResponseField name="connectionDate" type="string">Date the two profiles connected (`YYYY-MM-DD HH:MM:SS`). Empty unless `relationship` is `"1st degree"`.</ResponseField>
    <ResponseField name="email" type="string">Email from the contact info section. Only visible for 1st degree connections that share it, otherwise `""`.</ResponseField>
    <ResponseField name="phone" type="string">Phone from the contact info section, same visibility as `email`.</ResponseField>
    <ResponseField name="website" type="string">Website from the contact info section, same visibility as `email`.</ResponseField>
    <ResponseField name="open_link" type="boolean">Reachable via Open Profile (free InMail). Only with `sales_nav: true`.</ResponseField>
    <ResponseField name="inmail_eligible" type="boolean">InMail can be sent (no restriction). Only with `sales_nav: true`.</ResponseField>
    <ResponseField name="inmail_restriction" type="string">Reason an InMail cannot be sent, `"NO_RESTRICTION"` otherwise. Only with `sales_nav: true`.</ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="metadata" type="object">
  <Expandable title="metadata properties">
    <ResponseField name="action" type="string">
      The action that was performed (`"get"`).
    </ResponseField>
    <ResponseField name="account_id" type="string">
      The account ID used for the request.
    </ResponseField>
    <ResponseField name="credits_consumed" type="integer">
      Number of credits consumed (1).
    </ResponseField>
    <ResponseField name="timestamp" type="string">
      ISO 8601 timestamp of the response.
    </ResponseField>
  </Expandable>
</ResponseField>
