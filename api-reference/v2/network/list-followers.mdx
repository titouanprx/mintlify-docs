---

## Overview

Returns the account's followers (`list_followers`) or the people it follows (`list_following`), in LinkedIn's order. Followed company pages are not included.

<Note>This endpoint consumes **1 credit per 10 results** returned. Ignore this if you're on per-seat pricing: usage is unlimited.</Note>

## Request

<ParamField header="x-api-key" type="string" required>
  Your API key for authentication.
</ParamField>

<ParamField body="account_id" type="string" required>
  The unique identifier of the account to use.
</ParamField>

<ParamField body="action" type="string" required>
  `"list_followers"` or `"list_following"`.
</ParamField>

<ParamField body="params" type="object">
  <Expandable title="params properties">
    <ParamField body="count" type="integer" default={20}>
      Number of people to return. **Capped at `100`** per call — to retrieve more, paginate with `offset`.
    </ParamField>
    <ParamField body="offset" type="integer" default={0}>
      Zero-indexed offset into the list. Use `pagination.next_offset` from a previous response to fetch the next page.
    </ParamField>
  </Expandable>
</ParamField>

## Response

<ResponseField name="data" type="object">
  <Expandable title="data properties">
    <ResponseField name="followers" type="array">
      `list_followers` only. `list_following` returns the same objects under `following`.
      <Expandable title="person properties">
        <ResponseField name="name" type="string">Display name.</ResponseField>
        <ResponseField name="job_title" type="string">Headline.</ResponseField>
        <ResponseField name="profile_url" type="string">Public profile URL.</ResponseField>
        <ResponseField name="profile_urn" type="string">Member URN (`urn:li:fsd_profile:ACoA…`).</ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="total_available_results" type="integer">
      Total number of followers (or followed people). Returned when `offset` is `0`, `null` otherwise.
    </ResponseField>
    <ResponseField name="pagination" type="object">
      Same object as [`list_connections`](/api-reference/v2/network/list-connections): `offset`, `count`, `next_offset`, `has_more`.
    </ResponseField>
  </Expandable>
</ResponseField>
