---
title: 'Unfollow'
api: 'POST /v2/network'
description: 'Unfollow a member profile or a company page.'
---

## Overview

Stops following a person or a company page. Pass `profile_url` for a person, `company_url` for a page. Unfollowing a connection keeps the connection; only their posts leave the feed.

<Note>This endpoint consumes **1 credit** per request. Ignore this if you're on per-seat pricing: usage is unlimited.</Note>

## Request

<ParamField header="x-api-key" type="string" required>
  Your API key for authentication.
</ParamField>

<ParamField body="account_id" type="string" required>
  The unique identifier of the account to use.
</ParamField>

<ParamField body="action" type="string" required>
  Must be `"unfollow"`.
</ParamField>

<ParamField body="params" type="object" required>
  <Expandable title="params properties">
    <ParamField body="profile_url" type="string">
      The person to unfollow: profile URL (`https://www.linkedin.com/in/john-doe/`), public slug (`john-doe`), member id (`ACoAA…`) or `urn:li:fsd_profile:…`. **Required to unfollow a person.**
    </ParamField>
    <ParamField body="company_url" type="string">
      The company page to unfollow: URL (`https://www.linkedin.com/company/google/`) or slug (`google`). **Required to unfollow a page, unless `company_id` is given.**
    </ParamField>
    <ParamField body="company_id" type="string">
      Numeric LinkedIn company id, as an alternative to `company_url`.
    </ParamField>
  </Expandable>
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Indicates whether the request was successful.
</ResponseField>

<ResponseField name="data" type="object">
  <Expandable title="data properties">
    <ResponseField name="following" type="boolean">
      `false` once LinkedIn confirms the account no longer follows the target.
    </ResponseField>
    <ResponseField name="profile_id" type="string">
      Member id of the unfollowed person (`ACoAA…`). Person only.
    </ResponseField>
    <ResponseField name="profile_urn" type="string">
      `urn:li:fsd_profile:…` of the unfollowed person. Person only.
    </ResponseField>
    <ResponseField name="follower_count" type="integer">
      The person's follower count after the action, when LinkedIn reports it. Person only.
    </ResponseField>
    <ResponseField name="company_id" type="string">
      Numeric id of the unfollowed company. Page only.
    </ResponseField>
    <ResponseField name="company_urn" type="string">
      `urn:li:fsd_company:…` of the unfollowed company. Page only.
    </ResponseField>
    <ResponseField name="name" type="string">
      Company name. Page only.
    </ResponseField>
    <ResponseField name="universal_name" type="string">
      Company slug. Page only.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="metadata" type="object">
  <Expandable title="metadata properties">
    <ResponseField name="action" type="string">The action that was performed (`"unfollow"`).</ResponseField>
    <ResponseField name="account_id" type="string">The account ID used for the request.</ResponseField>
    <ResponseField name="credits_consumed" type="integer">Number of credits consumed (1).</ResponseField>
    <ResponseField name="timestamp" type="string">ISO 8601 timestamp of the response.</ResponseField>
  </Expandable>
</ResponseField>
